    <div class="modal-title" id="modal-title">VALID TICKET</div>
    <div class="modal-detail" id="modal-detail">Code: FST-000000</div>
    <button class="btn-reset" onclick="resetScanner()">Scan Next Ticket</button>

  <script>
    // REPLACE THESE WITH YOUR SUPABASE CREDENTIALS FROM STEP 1
    const SUPABASE_URL = "YOUR_SUPABASE_URL_HERE";
    const SUPABASE_KEY = "YOUR_SUPABASE_ANON_KEY_HERE";
    const db = supabase.createClient(SUPABASE_URL, SUPABASE_KEY);

    let html5QrcodeScanner;

    async function onScanSuccess(decodedText) {
      html5QrcodeScanner.pause();
      const code = decodedText.trim();

      // Check database for ticket
      const { data: ticket, error } = await db
        .from('tickets')
        .select('*')
        .eq('ticket_code', code)
        .single();

      if (error || !ticket) {
        showModal('invalid', '❌ INVALID TICKET', `Code: ${code}<br>Ticket not found in database!`);
      } else if (ticket.status === 'CHECKED IN') {
        showModal('invalid', '⚠️️ ALREADY SCANNED!', `Code: ${code}<br>Scanned at: ${new Date(ticket.scanned_at).toLocaleTimeString()}`);
      } else {
        // Mark ticket as CHECKED IN
        await db.from('tickets').update({
          status: 'CHECKED IN',
          scanned_at: new Date().toISOString(),
          gatekeeper: 'Gate 1'
        }).eq('ticket_code', code);

        showModal('valid', '✅ VALID TICKET', `Code: ${code}<br>Type: ${ticket.ticket_type}`);
      }
    }

    function showModal(type, title, detail) {
      const modal = document.getElementById('result-modal');
      modal.className = type;
      document.getElementById('modal-title').innerHTML = title;
      document.getElementById('modal-detail').innerHTML = detail;
      modal.style.display = 'flex';
    }

    function resetScanner() {
      document.getElementById('result-modal').style.display = 'none';
      html5QrcodeScanner.resume();
    }

    window.onload = () => {
      html5QrcodeScanner = new Html5QrcodeScanner("reader", { fps: 10, qrbox: { width: 250, height: 250 } });
      html5QrcodeScanner.render(onScanSuccess);
    };
  </script>
